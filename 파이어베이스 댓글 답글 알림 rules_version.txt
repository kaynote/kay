rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() &&
        request.auth.uid == "eEltgLaV6oN7MHUXTfQONc2wGAk1";
    }

    /* =====================
       USERS
    ===================== */

    match /users/{uid} {

      // 전체 사용자 목록은 관리자만 조회 가능
      allow read: if isAdmin();

      // 로그인한 사용자는 자기 문서만 생성
      allow create: if isSignedIn()
        && request.auth.uid == uid;

      // 로그인한 사용자는 자기 문서만 수정
      allow update: if isSignedIn()
        && request.auth.uid == uid;

      // 삭제는 관리자만 가능
      allow delete: if isAdmin();
    }

    /* =====================
       NOTIFICATIONS
    ===================== */

    match /notifications/{id} {

      // CREATE
      allow create: if isSignedIn()
        && request.resource.data.senderUid == request.auth.uid
        && request.resource.data.targetUid is string
        && request.resource.data.targetUid != request.auth.uid
        && request.resource.data.read == false;

      // READ (본인만)
      allow read: if isSignedIn()
        && resource.data.targetUid == request.auth.uid;

      // UPDATE (read만 변경)
      allow update: if isSignedIn()
        && resource.data.targetUid == request.auth.uid
        && request.resource.data.diff(resource.data)
            .affectedKeys()
            .hasOnly(["read"]);

      // DELETE
      allow delete: if isSignedIn()
        && (
          resource.data.targetUid == request.auth.uid
          || isAdmin()
        );
    }

    /* =====================
       PEOPLE (POSTS)
    ===================== */

    match /people/{postId} {

      allow read: if true;

      /* -------- COMMENTS -------- */

      match /comments/{commentId} {

        allow read: if true;

        allow create: if isSignedIn()
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.text is string
          && request.resource.data.text.size() <= 1000
          && request.resource.data.createdAt == request.time
          && request.resource.data.keys().hasOnly([
                "uid",
                "name",
                "photo",
                "text",
                "createdAt",
                "parentId",
                "editedAt",
                "deleted",
                "originalText",
                "deletedAt"
              ]);

        allow update: if isSignedIn()
          && (resource.data.uid == request.auth.uid || isAdmin())
          && request.resource.data.uid == resource.data.uid
          && request.resource.data.name == resource.data.name
          && request.resource.data.photo == resource.data.photo
          && request.resource.data.createdAt == resource.data.createdAt
          && request.resource.data.text is string
          && request.resource.data.text.size() <= 1000;

        allow delete: if isSignedIn()
          && (resource.data.uid == request.auth.uid || isAdmin());
      }

      /* -------- LIKES -------- */

      match /likes/{uid} {

        allow read: if true;

        allow create: if isSignedIn()
          && request.auth.uid == uid;

        allow delete: if isSignedIn()
          && request.auth.uid == uid;
      }
    }
  }
}
