rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    /* =====================
       AUTH HELPERS
    ===================== */

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() &&
        request.auth.uid == "eEltgLaV6oN7MHUXTfQONc2wGAk1";
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    function isValidString(field, maxSize) {
      return field is string && field.size() > 0 && field.size() <= maxSize;
    }

    /* =====================
       NOTIFICATIONS (SECURE)
    ===================== */

    match /notifications/{id} {

      // CREATE: senderUid 반드시 본인이어야 함 (조작 방지 핵심)
      allow create: if isSignedIn()
        && request.resource.data.senderUid == request.auth.uid
        && isValidString(request.resource.data.targetUid, 128)
        && isValidString(request.resource.data.commentId, 128)
        && isValidString(request.resource.data.type, 20)
        && request.resource.data.read == false;

      // READ: 본인 or 관리자만
      allow read: if isSignedIn()
        && (resource.data.targetUid == request.auth.uid || isAdmin());

      // UPDATE: read 상태만 변경 가능 (나머지 필드 보호)
      allow update: if isSignedIn()
        && (resource.data.targetUid == request.auth.uid || isAdmin())
        && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(["read"]);

      // DELETE: 관리자만 (유저 삭제 차단)
      allow delete: if isAdmin();
    }

    /* =====================
       PEOPLE (POSTS)
    ===================== */

    match /people/{postId} {

      allow read: if true;

      /* =====================
         COMMENTS
      ===================== */

      match /comments/{commentId} {

        allow read: if true;

        allow create: if isSignedIn()
          && request.resource.data.uid == request.auth.uid
          && isValidString(request.resource.data.text, 1000)
          && (request.resource.data.parentId == null
              || request.resource.data.parentId is string);

        allow update: if isSignedIn()
          && (resource.data.uid == request.auth.uid || isAdmin());

        allow delete: if isSignedIn()
          && (resource.data.uid == request.auth.uid || isAdmin());
      }

      /* =====================
         LIKES
      ===================== */

      match /likes/{uid} {

        allow read: if true;

        allow create: if isSignedIn()
          && request.auth.uid == uid;

        allow delete: if isSignedIn()
          && request.auth.uid == uid;
      }
    }
  }
}
